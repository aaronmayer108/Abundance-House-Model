<script>

const fmt = n => n >= 1e9 ? `$${(n/1e9).toFixed(2)}B` : n >= 1e6 ? `$${(n/1e6).toFixed(1)}M` : n >= 1e3 ? `$${(n/1e3).toFixed(0)}K` : `$${Math.round(n)}`;


let scenarios = [{name:'Scenario A'},{name:'Scenario B'}];

let activeIdx = 0;

let states = [];


function defaultState() {

  return { duration:2, equity:2, founders:15, residencies:4, outcome:100, prob:15, fanciness:25, personnel:100, discount:80, dilution:20 };

}


function calcDist(prob) {

  const shift = (prob - 15) / 15;

  let great_w = Math.max(0.1, Math.min(15, 1 + shift * 2));

  let good_w  = Math.max(1,   Math.min(30, 9 + shift * 4));

  let fail_w  = Math.max(0, 100 - great_w - good_w);

  const tot = fail_w + good_w + great_w;

  return { fail: fail_w/tot*100, good: good_w/tot*100, great: great_w/tot*100 };

}


function calcMetrics(s) {

  const maxRes = s.duration===3 ? 4 : s.duration===2 ? 6 : 12;

  const res = Math.min(s.residencies, maxRes);

  const dist = calcDist(s.prob);

  const ev = (dist.fail/100)*1 + (dist.good/100)*10 + (dist.great/100)*100;

  const valuePerFounder = s.outcome*1e6 * ev * (s.equity/100) * (1 - s.dilution/100) * (1 - s.discount/100);

  const equityPerBatch = s.founders * (s.equity/100);

  const batchValue = s.founders * valuePerFounder;

  const totalValue = res * batchValue;

  const baseCost = 25000 + (s.founders - 12) * 2000;

  const costPerMonth = Math.max(5000, baseCost) * (s.fanciness / 25);

  const costPerResidency = costPerMonth * s.duration + (s.personnel * 1000 / res);

  const totalCost = costPerResidency * res;

  return { res, dist, ev, equityPerBatch, batchValue, totalValue, costPerResidency, totalCost };

}


function maxResForDur(dur) { return dur===3 ? 4 : dur===2 ? 6 : 12; }


function distPills(dist) {

  return `<div class="dist-pill pill-fail"><div class="pill-label">Flat/fail</div>${dist.fail.toFixed(0)}%</div>

          <div class="dist-pill pill-good"><div class="pill-label">10× outcome</div>${dist.good.toFixed(0)}%</div>

          <div class="dist-pill pill-great"><div class="pill-label">100× outcome</div>${dist.great.toFixed(0)}%</div>`;

}


function buildMetrics(m, scName) {

  return `

    <div class="metric"><div class="metric-label">Total portfolio value</div><div class="metric-value">${fmt(m.totalValue)}</div><div class="metric-sub">${m.res} residencies · post-dilution, NPV-adjusted</div></div>

    <div class="metric"><div class="metric-label">Value per batch</div><div class="metric-value">${fmt(m.batchValue)}</div><div class="metric-sub">Gross equity per cohort</div></div>

    <div class="metric"><div class="metric-label">Cost per residency</div><div class="metric-value">${fmt(m.costPerResidency)}</div><div class="metric-sub">Incl. personnel allocation</div></div>

    <div class="metric"><div class="metric-label">Total annual cost</div><div class="metric-value">${fmt(m.totalCost)}</div><div class="metric-sub">All residencies + personnel</div></div>

    <div class="metric"><div class="metric-label">Equity per batch</div><div class="metric-value">${m.equityPerBatch.toFixed(1)}%</div><div class="metric-sub">${m.res} batches/yr</div></div>

    <div class="metric"><div class="metric-label">EV multiple</div><div class="metric-value">${m.ev.toFixed(2)}×</div><div class="metric-sub">Probability-weighted return</div></div>`;

}


function knobHTML(key, label, val, min, max, step, fmtFn) {

  return `<div class="knob">

    <div class="knob-header">

      <span class="knob-label">${label}</span>

      <span class="knob-value" id="kv-${key}">${fmtFn(val)}</span>

    </div>

    <input type="range" min="${min}" max="${max}" step="${step}" value="${val}" data-key="${key}" data-fmt-id="${key}">

  </div>`;

}


function buildScenario(s, m, i) {

  const mr = maxResForDur(s.duration);

  return `

    <p class="section-title">Structure</p>

    <div class="knobs">

      ${knobHTML('duration','Duration (months)',s.duration,1,3,1,v=>v+' mo')}

      ${knobHTML('equity','Equity charge',s.equity,0.1,5,0.1,v=>v+'%')}

      ${knobHTML('founders','Founders per batch',s.founders,8,40,1,v=>v+' founders')}

      ${knobHTML('residencies','Residencies per year',Math.min(s.residencies,mr),1,mr,1,v=>v+'/yr')}

    </div>

    <p class="section-title">Expected outcome per founder</p>

    <div class="knobs">

      ${knobHTML('outcome','Expected exit value',s.outcome,10,1000,10,v=>'$'+v+'M')}

      ${knobHTML('prob','Probability / success tilt',s.prob,1,50,1,v=>v+'%')}

    </div>

    <div class="dist-row" id="dist-${i}">${distPills(m.dist)}</div>

    <p class="section-title">Costs</p>

    <div class="knobs">

      ${knobHTML('fanciness','Fanciness ($/month baseline)',s.fanciness,25,100,5,v=>'$'+v+'k/mo')}

      ${knobHTML('personnel','Personnel cost',s.personnel,0,500,25,v=>'$'+v+'k/yr')}

    </div>

    <p class="section-title">Value adjustments</p>

    <div class="knobs">

      ${knobHTML('discount','Discount rate (NPV haircut)',s.discount,0,100,1,v=>v+'%')}

      ${knobHTML('dilution','Dilution assumption',s.dilution,0,100,1,v=>v+'%')}

    </div>

    <p class="section-title">Outputs</p>

    <div class="metrics" id="metrics-${i}">${buildMetrics(m)}</div>`;

}


const fmtFns = {

  duration:v=>v+' mo', equity:v=>v+'%', founders:v=>v+' founders',

  residencies:v=>v+'/yr', outcome:v=>'$'+v+'M', prob:v=>v+'%',

  fanciness:v=>'$'+v+'k/mo', personnel:v=>'$'+v+'k/yr',

  discount:v=>v+'%', dilution:v=>v+'%'

};


function renderAll() {

  const tabsEl = document.getElementById('tabs');

  const scenEl = document.getElementById('scenarios');

  tabsEl.innerHTML = '';

  scenEl.innerHTML = '';


  scenarios.forEach((sc, i) => {

    if (!states[i]) states[i] = defaultState();

    const btn = document.createElement('button');

    btn.className = 'tab-btn' + (i === activeIdx ? ' active' : '');

    btn.textContent = sc.name;

    btn.onclick = () => { activeIdx = i; renderAll(); };

    tabsEl.appendChild(btn);

  });


  if (scenarios.length < 4) {

    const add = document.createElement('button');

    add.className = 'tab-btn add-btn';

    add.innerHTML = '+ Add scenario';

    add.onclick = () => {

      const letters = ['A','B','C','D'];

      scenarios.push({ name: 'Scenario ' + letters[scenarios.length] });

      states.push(defaultState());

      activeIdx = scenarios.length - 1;

      renderAll();

    };

    tabsEl.appendChild(add);

  }


  scenarios.forEach((sc, i) => {

    const s = states[i];

    const m = calcMetrics(s);

    const div = document.createElement('div');

    div.className = 'scenario' + (i === activeIdx ? ' active' : '');

    div.innerHTML = buildScenario(s, m, i);

    scenEl.appendChild(div);


    div.querySelectorAll('input[type=range]').forEach(inp => {

      inp.addEventListener('input', () => {

        const key = inp.dataset.key;

        states[i][key] = parseFloat(inp.value);

        if (key === 'duration') {

          const mr = maxResForDur(states[i].duration);

          if (states[i].residencies > mr) states[i].residencies = mr;

        }

        updateScenario(i);

        updateCompare();

      });

    });

  });


  updateCompare();

}


function updateScenario(i) {

  const s = states[i];

  const m = calcMetrics(s);

  const sc = document.querySelectorAll('.scenario')[i];

  if (!sc) return;

  sc.querySelectorAll('input[type=range]').forEach(inp => {

    const key = inp.dataset.key;

    const mr = maxResForDur(s.duration);

    if (key === 'residencies') inp.max = mr;

    inp.value = s[key];

    const span = inp.previousElementSibling?.querySelector('.knob-value');

    if (span && fmtFns[key]) span.textContent = fmtFns[key](s[key]);

  });

  const distEl = sc.querySelector('[id^="dist-"]');

  if (distEl) distEl.innerHTML = distPills(m.dist);

  const metricsEl = sc.querySelector('[id^="metrics-"]');

  if (metricsEl) metricsEl.innerHTML = buildMetrics(m);

}


function updateCompare() {

  document.getElementById('compare').innerHTML = scenarios.map((sc, i) => {

    const m = calcMetrics(states[i] || defaultState());

    return `<div class="compare-card">

      <div class="compare-card-title">${sc.name}</div>

      <div class="compare-metric"><span class="compare-key">Portfolio value</span><span class="compare-val">${fmt(m.totalValue)}</span></div>

      <div class="compare-metric"><span class="compare-key">Per-batch value</span><span class="compare-val">${fmt(m.batchValue)}</span></div>

      <div class="compare-metric"><span class="compare-key">Annual cost</span><span class="compare-val">${fmt(m.totalCost)}</span></div>

      <div class="compare-metric"><span class="compare-key">Equity/batch</span><span class="compare-val">${m.equityPerBatch.toFixed(1)}%</span></div>

      <div class="compare-metric"><span class="compare-key">EV multiple</span><span class="compare-val">${m.ev.toFixed(2)}×</span></div>

    </div>`;

  }).join('');

}


renderAll();

</script>
